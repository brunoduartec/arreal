% Iterate through possible chapter files but only input when the file exists.
% This prevents compilation errors when some chapter files are intentionally missing.
% You can optionally define a comma-separated list of chapter names
% (for example: \def\chapterslist{0,1,2-3,4,5}) to control which files
% are included and to allow merged chapter files like `chapter2-3.tex`.
% If \chapterslist is not defined, the code falls back to the numeric
% sequence 0..44 and will include `chapters/chapter<NUM>.tex` when present.
\makeatletter
\@ifundefined{chapterslist}{%
	% Default behaviour: iterate numeric chapters 0..44 and input when file exists
	\foreach \chapternumber in {0,...,44}{%
		\IfFileExists{chapters/chapter\chapternumber.tex}{%
			\input{chapters/chapter\chapternumber.tex}%
		}{}%
	}
}{%
	% User provided a custom comma-separated list in \chapterslist
	% Example: \def\chapterslist{0,1,2-3,4}
	\@for\ch:=\chapterslist\do{%
		\IfFileExists{chapters/chapter\ch.tex}{%
			\input{chapters/chapter\ch.tex}%
		}{}%
	}
}
\makeatother
